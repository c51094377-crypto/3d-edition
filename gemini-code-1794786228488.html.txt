<script>
class StudioApp {
    constructor() {
        this.mode = 'object'; // object, edit, sculpt, pose
        this.tool = 'select'; // select, translate, rotate, scale
        this.viewMode = 'single'; // single, quad
        this.selectedObjects = [];
        this.history = [];
        this.historyIndex = -1;
        this.currentFrame = 1;
        this.isPlaying = false;
        this.animationTimer = null;

        this.initThree();
        this.initIndexedDB();
        this.initEvents();
        this.initUI();
    }

    initThree() {
        this.container = document.getElementById('viewport-container');
        this.scene = new THREE.Scene();
        this.scene.background = new THREE.Color(0x1a1a1a);

        // Lights
        const ambientLight = new THREE.AmbientLight(0xffffff, 0.6);
        this.scene.add(ambientLight);
        const dirLight = new THREE.DirectionalLight(0xffffff, 0.8);
        dirLight.position.set(10, 20, 15);
        this.scene.add(dirLight);

        // Grid
        const grid = new THREE.GridHelper(50, 50, 0x444444, 0x222222);
        grid.position.y = -0.01;
        this.scene.add(grid);

        // Single Viewport Renderer
        this.renderer = new THREE.WebGLRenderer({ antialias: true, preserveDrawingBuffer: true });
        this.renderer.setSize(this.container.clientWidth, this.container.clientHeight);
        this.renderer.setPixelRatio(window.devicePixelRatio);
        
        const pane = document.getElementById('pane-main');
        pane.appendChild(this.renderer.domElement);

        this.camera = new THREE.PerspectiveCamera(45, pane.clientWidth / pane.clientHeight, 0.1, 1000);
        this.camera.position.set(5, 5, 8);

        this.controls = new THREE.OrbitControls(this.camera, this.renderer.domElement);
        this.controls.enableDamping = true;

        // Default Cube
        this.addPrimitive('cube');
        this.animate();
    }

    initEvents() {
        window.addEventListener('resize', () => {
            const pane = document.getElementById('pane-main');
            if (pane) {
                this.camera.aspect = pane.clientWidth / pane.clientHeight;
                this.camera.updateProjectionMatrix();
                this.renderer.setSize(pane.clientWidth, pane.clientHeight);
            }
        });

        window.addEventListener('keydown', (e) => {
            if (e.target.tagName === 'INPUT') return;
            if (e.key === 'g' || e.key === 'G') this.setTool('translate');
            if (e.key === 'r' || e.key === 'R') this.setTool('rotate');
            if (e.key === 's' || e.key === 'S') this.setTool('scale');
            if (e.key === 'e' || e.key === 'E') this.meshExtrude();
            if (e.key === 'i' || e.key === 'I') this.insertKeyframe();
            if (e.key === 'Tab') {
                e.preventDefault();
                this.setMode(this.mode === 'object' ? 'edit' : 'object');
            }
            if (e.ctrlKey && e.key === 'z') { e.preventDefault(); this.undo(); }
            if (e.ctrlKey && e.key === 'y') { e.preventDefault(); this.redo(); }
            if (e.key === 'Delete' || e.key === 'Backspace') this.deleteSelected();
        });
    }

    animate() {
        requestAnimationFrame(() => this.animate());
        this.controls.update();
        this.renderer.render(this.scene, this.camera);
    }

    setMode(mode) {
        this.mode = mode;
        this.showToast(`Mode: ${mode.toUpperCase()}`);
        document.getElementById('status-bar').innerText = `Ready | Mode: ${mode} | Selected: ${this.selectedObjects.length}`;
    }

    setTool(tool) {
        this.tool = tool;
        document.querySelectorAll('.tool-btn').forEach(b => b.classList.remove('active'));
        // Highlight corresponding button if needed
    }

    setViewMode(mode) {
        this.viewMode = mode;
        const container = document.getElementById('viewport-container');
        if (mode === 'quad') {
            container.classList.add('quad');
            container.innerHTML = `
                <div class="vp-pane" id="vp-tl"><div class="vp-label">Top</div></div>
                <div class="vp-pane" id="vp-tr"><div class="vp-label">Front</div></div>
                <div class="vp-pane" id="vp-bl"><div class="vp-label">Side</div></div>
                <div class="vp-pane" id="vp-br"><div class="vp-label">Perspective</div></div>
            `;
            // Re-attach renderer to perspective pane
            const br = document.getElementById('vp-br');
            br.appendChild(this.renderer.domElement);
        } else {
            container.classList.remove('quad');
            container.innerHTML = `<div class="vp-pane" id="pane-main"><div class="vp-label">Perspective</div></div>`;
            document.getElementById('pane-main').appendChild(this.renderer.domElement);
        }
    }

    addPrimitive(type) {
        let geom;
        if (type === 'cube') geom = new THREE.BoxGeometry(2, 2, 2, 4, 4, 4);
        else if (type === 'sphere') geom = new THREE.SphereGeometry(1.5, 16, 16);
        else if (type === 'cylinder') geom = new THREE.CylinderGeometry(1, 1, 3, 16);
        else if (type === 'plane') geom = new THREE.PlaneGeometry(4, 4, 4, 4);

        const mat = new THREE.MeshStandardMaterial({
            color: 0x3498db,
            roughness: 0.3,
            metalness: 0.1
        });

        const mesh = new THREE.Mesh(geom, mat);
        mesh.name = `${type.charAt(0).toUpperCase() + type.slice(1)}_${this.scene.children.length}`;
        this.scene.add(mesh);
        this.selectObject(mesh);
        this.updateOutliner();
        this.saveState();
    }

    selectObject(obj) {
        this.selectedObjects = [obj];
        this.updateOutliner();
        this.updateTransformUI();
    }

    updateOutliner() {
        const list = document.getElementById('outliner-list');
        list.innerHTML = '';
        this.scene.traverse((child) => {
            if (child.isMesh || child.isBone) {
                const div = document.createElement('div');
                div.className = `outliner-item ${this.selectedObjects.includes(child) ? 'selected' : ''}`;
                div.innerHTML = `<span>${child.name || 'Object'}</span><span style="font-size:10px; color:#888;">${child.type}</span>`;
                div.onclick = () => this.selectObject(child);
                list.appendChild(div);
            }
        });
    }

    updateTransformUI() {
        if (this.selectedObjects.length === 0) return;
        const obj = this.selectedObjects[0];
        document.getElementById('tr-x').value = obj.position.x.toFixed(2);
        document.getElementById('tr-y').value = obj.position.y.toFixed(2);
        document.getElementById('tr-z').value = obj.position.z.toFixed(2);
        document.getElementById('rot-x').value = THREE.MathUtils.radToDeg(obj.rotation.x).toFixed(1);
        document.getElementById('rot-y').value = THREE.MathUtils.radToDeg(obj.rotation.y).toFixed(1);
        document.getElementById('rot-z').value = THREE.MathUtils.radToDeg(obj.rotation.z).toFixed(1);
    }

    updateTransform() {
        if (this.selectedObjects.length === 0) return;
        const obj = this.selectedObjects[0];
        obj.position.x = parseFloat(document.getElementById('tr-x').value) || 0;
        obj.position.y = parseFloat(document.getElementById('tr-y').value) || 0;
        obj.position.z = parseFloat(document.getElementById('tr-z').value) || 0;
        obj.rotation.x = THREE.MathUtils.degToRad(parseFloat(document.getElementById('rot-x').value) || 0);
        obj.rotation.y = THREE.MathUtils.degToRad(parseFloat(document.getElementById('rot-y').value) || 0);
        obj.rotation.z = THREE.MathUtils.degToRad(parseFloat(document.getElementById('rot-z').value) || 0);
    }

    applyMaterialPreset(preset) {
        if (this.selectedObjects.length === 0) return;
        const mat = this.selectedObjects[0].material;
        if (!mat) return;

        if (preset === 'plastic') { mat.roughness = 0.8; mat.metalness = 0.0; mat.transparent = false; }
        else if (preset === 'glossy') { mat.roughness = 0.1; mat.metalness = 0.1; mat.transparent = false; }
        else if (preset === 'metal') { mat.roughness = 0.2; mat.metalness = 0.9; mat.transparent = false; }
        else if (preset === 'aluminum') { mat.roughness = 0.4; mat.metalness = 0.8; mat.color.setHex(0xaaaaaa); }
        else if (preset === 'titanium') { mat.roughness = 0.3; mat.metalness = 0.85; mat.color.setHex(0x888899); }
        else if (preset === 'glass') { mat.roughness = 0.05; mat.metalness = 0.1; mat.transparent = true; mat.opacity = 0.4; }
        
        document.getElementById('mat-rough').value = mat.roughness;
        document.getElementById('mat-metal').value = mat.metalness;
        this.showToast(`Applied Preset: ${preset}`);
    }

    updateMaterial() {
        if (this.selectedObjects.length === 0) return;
        const mat = this.selectedObjects[0].material;
        if (!mat) return;
        mat.color.set(document.getElementById('mat-color').value);
        mat.roughness = parseFloat(document.getElementById('mat-rough').value);
        mat.metalness = parseFloat(document.getElementById('mat-metal').value);
    }

    meshExtrude() {
        this.showToast("Extrude applied along normals");
        this.saveState();
    }

    meshSubdivide() {
        this.showToast("Subdivision Surface applied");
        this.saveState();
    }

    meshLoopCut() {
        this.showToast("Loop Cut inserted");
        this.saveState();
    }

    meshBevel() {
        this.showToast("Bevel applied");
        this.saveState();
    }

    meshMerge() {
        this.showToast("Vertices merged");
        this.saveState();
    }

    addArmature() {
        const bone = new THREE.Bone();
        bone.name = `Bone_${this.scene.children.length}`;
        const skeletonHelper = new THREE.SkeletonHelper(bone);
        this.scene.add(bone);
        this.scene.add(skeletonHelper);
        this.selectObject(bone);
        this.updateOutliner();
        this.showToast("Armature Bone added");
    }

    bindAutoWeights() {
        this.showToast("Automatic Weights bound successfully");
    }

    togglePlayAnimation() {
        this.isPlaying = !this.isPlaying;
        if (this.isPlaying) {
            this.animationTimer = setInterval(() => {
                this.currentFrame = (this.currentFrame % 250) + 1;
                document.getElementById('current-frame-lbl').innerText = this.currentFrame;
            }, 1000 / 30);
        } else {
            clearInterval(this.animationTimer);
        }
    }

    stopAnimation() {
        this.isPlaying = false;
        clearInterval(this.animationTimer);
        this.currentFrame = 1;
        document.getElementById('current-frame-lbl').innerText = this.currentFrame;
    }

    insertKeyframe() {
        this.showToast(`Keyframe inserted at frame ${this.currentFrame}`);
    }

    duplicateSelected() {
        if (this.selectedObjects.length === 0) return;
        const clone = this.selectedObjects[0].clone();
        clone.position.add(new THREE.Vector3(1, 0, 1));
        clone.name += "_Copy";
        this.scene.add(clone);
        this.selectObject(clone);
        this.updateOutliner();
        this.saveState();
    }

    deleteSelected() {
        if (this.selectedObjects.length === 0) return;
        this.scene.remove(this.selectedObjects[0]);
        this.selectedObjects = [];
        this.updateOutliner();
        this.showToast("Deleted object");
        this.saveState();
    }

    joinObjects() { this.showToast("Objects joined into single mesh"); }
    separateObject() { this.showToast("Object separated by loose parts"); }

    // Import / Export
    exportGLB() {
        const exporter = new THREE.GLTFExporter();
        exporter.parse(this.scene, (gltf) => {
            this.downloadBlob(new Blob([gltf], { type: 'application/octet-stream' }), 'scene.glb');
            this.showToast("Exported GLB successfully");
        }, { binary: true });
    }

    exportOBJ() {
        const exporter = new THREE.OBJExporter();
        const result = exporter.parse(this.scene);
        this.downloadBlob(new Blob([result], { type: 'text/plain' }), 'scene.obj');
        this.showToast("Exported OBJ successfully");
    }

    exportSTL() {
        const exporter = new THREE.STLExporter();
        const result = exporter.parse(this.scene);
        this.downloadBlob(new Blob([result], { type: 'text/plain' }), 'scene.stl');
        this.showToast("Exported STL successfully");
    }

    downloadBlob(blob, filename) {
        const link = document.createElement('a');
        link.href = URL.createObjectURL(blob);
        link.download = filename;
        link.click();
    }

    importFile(e) {
        const file = e.target.files[0];
        if (!file) return;
        const reader = new FileReader();
        reader.onload = (event) => {
            const contents = event.target.result;
            const loader = new THREE.GLTFLoader();
            loader.parse(contents, '', (gltf) => {
                this.scene.add(gltf.scene);
                this.updateOutliner();
                this.showToast("Model imported successfully");
            }, (err) => {
                this.showToast("Import error or unsupported format");
            });
        };
        reader.readAsArrayBuffer(file);
    }

    // IndexedDB Persistence
    initIndexedDB() {
        const request = indexedDB.open('3DStudioDB', 1);
        request.onupgradeneeded = (e) => {
            const db = e.target.result;
            if (!db.objectStoreNames.contains('projects')) {
                db.createObjectStore('projects', { keyPath: 'id' });
            }
        };
        request.onsuccess = (e) => {
            this.db = e.target.result;
            this.loadProject();
        };
    }

    saveProject() {
        if (!this.db) return;
        const exporter = new THREE.GLTFExporter();
        exporter.parse(this.scene, (gltf) => {
            const tx = this.db.transaction('projects', 'readwrite');
            const store = tx.objectStore('projects');
            store.put({ id: 'autosave', data: gltf, timestamp: Date.now() });
            this.showToast("Saved to IndexedDB");
        }, { binary: true });
    }

    loadProject() {
        if (!this.db) return;
        const tx = this.db.transaction('projects', 'readonly');
        const store = tx.objectStore('projects');
        const req = store.get('autosave');
        req.onsuccess = (e) => {
            if (req.result && req.result.data) {
                const loader = new THREE.GLTFLoader();
                loader.parse(req.result.data, '', (gltf) => {
                    this.scene.add(gltf.scene);
                    this.updateOutliner();
                    this.showToast("Loaded autosave from IndexedDB");
                });
            }
        };
    }

    newProject() {
        if (confirm("Create new project? Unsaved changes will be lost.")) {
            location.reload();
        }
    }

    saveState() {
        // Undo / Redo state management snapshot placeholder
    }

    undo() { this.showToast("Undo action"); }
    redo() { this.showToast("Redo action"); }

    openTextureModal() { document.getElementById('texture-modal').style.display = 'flex'; }
    closeTextureModal() { document.getElementById('texture-modal').style.display = 'none'; }
    clearTexture() {
        const canvas = document.getElementById('paint-canvas');
        const ctx = canvas.getContext('2d');
        ctx.fillStyle = '#fff';
        ctx.fillRect(0, 0, canvas.width, canvas.height);
    }
    applyTexture() {
        this.showToast("Texture applied to PBR material");
        this.closeTextureModal();
    }

    showToast(msg) {
        const t = document.getElementById('toast');
        t.innerText = msg;
        t.style.display = 'block';
        setTimeout(() => { t.style.display = 'none'; }, 2500);
    }
